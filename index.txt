        // Tab Switching Logic
        function switchTab(tabId) {
            // Hide all tab contents
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            // Remove active style from all buttons
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active', 'bg-sbk-primary', 'text-white'));
            
            // Show selected page
            const targetPage = document.getElementById(`page-${tabId}`);
            if (targetPage) {
                targetPage.classList.remove('hidden');
            }

            // Highlight active tab button
            const activeTabBtn = document.getElementById(`tab-${tabId}`);
            if (activeTabBtn) {
                activeTabBtn.classList.add('active');
            }

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Submit and Score Quiz
        function submitQuiz() {
            let score = 0;

            currentActiveQuiz.forEach((item, idx) => {
                const ansBox = document.getElementById(`quiz-ans-${idx}`);
                ansBox.classList.remove('hidden', 'bg-emerald-100', 'text-emerald-800', 'bg-red-100', 'text-red-800');

                if (item.type === 'choice') {
                    const selected = document.querySelector(`input[name="quiz-${idx}"]:checked`);
                    if (selected && parseInt(selected.value) === item.ans) {
                        score++;
                        ansBox.classList.add('bg-emerald-100', 'text-emerald-800');
                        ansBox.innerHTML = `<i class="fa-solid fa-circle-check mr-1"></i> 정답입니다!`;
                    } else {
                        ansBox.classList.add('bg-red-100', 'text-red-800');
                        ansBox.innerHTML = `<i class="fa-solid fa-circle-xmark mr-1"></i> 오답! 정답: ${item.options[item.ans]}`;
                    }
                } else {
                    const inputVal = document.getElementById(`quiz-input-${idx}`).value.trim();
                    if (inputVal.toLowerCase() === item.ans.toLowerCase()) {
                        score++;
                        ansBox.classList.add('bg-emerald-100', 'text-emerald-800');
                        ansBox.innerHTML = `<i class="fa-solid fa-circle-check mr-1"></i> 정답입니다!`;
                    } else {
                        ansBox.classList.add('bg-red-100', 'text-red-800');
                        ansBox.innerHTML = `<i class="fa-solid fa-circle-xmark mr-1"></i> 오답! 정답: ${item.ans}`;
                    }
                }
            });

            const scoreBoard = document.getElementById('quiz-score-board');
            document.getElementById('score-text').innerText = `${score} / 25 점 (${Math.round(score/25*100)}점)`;
            scoreBoard.classList.remove('hidden');
            window.scrollTo({ top: scoreBoard.offsetTop - 100, behavior: 'smooth' });
        }

            element.click();
            document.body.removeChild(element);
        }
